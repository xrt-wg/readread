### 改动 1：ImportItemEditor.jsx

**a) 函数签名**：

```diff

```

**b) 新增导入**：

```diff
+import { getReading } from '../services/readings'
```

**c) 新增 useEffect（在现有 `useState` 块之后）**：

```js
// 按需加载完整数据（列表查询不含 sections），覆盖从列表传入的元数据
useEffect(() => {
  let active = true
  getReading(item.id, { canUseCloudLibrary, userId }).then(reading => {
    if (!active || !reading) return
    setTitle(reading.title)
    setAuthor(reading.author || '')
    setKind(reading.kind || 'article')
    setSourceUrl(reading.sourceUrl || '')
    setSections((reading.sections || []).map(s => ({
      id: s.id,
      heading: s.heading || '',
      bodyText: s.body?.text || '',
      bodyMarkdown: s.body?.markdown ?? null,
    })))
    snapshot.current = {
      title: reading.title || '',
      author: reading.author || '',
      kind: reading.kind || 'article',
      sourceUrl: reading.sourceUrl || '',
      sections: JSON.stringify((reading.sections || []).map(s => ({
        id: s.id, heading: s.heading || '',
        bodyText: s.body?.text || '', bodyMarkdown: s.body?.markdown ?? null,
      }))),
    }
  }).catch(() => {}) // 静默回退，列表数据兜底
  return () => { active = false }
}, [item.id, canUseCloudLibrary, userId])
```

## 有效性原理

```
打开编辑器
  ├─ 先用列表数据显示外壳（标题/作者等元数据立即可见）
  └─ useEffect 异步调用 getReading(item.id)
        └─ 内部使用 READING_DETAIL_COLUMNS（含 sections）
              └─ 返回完整数据 → setState 覆盖
                    ├─ sections 正文填充到内容编辑区
                    └─ snapshot 更新（脏检测基准）
```

`getReading` 对同 id + 同 auth 的重复请求会被 HTTP 缓存命中（Supabase 返回标准 Cache-Control 头），额外延迟几乎为零。

获取失败时 `.catch(() => {})` 静默回退——编辑器外壳已用列表数据显示标题等元数据，不影响用户继续编辑元数据字段。
