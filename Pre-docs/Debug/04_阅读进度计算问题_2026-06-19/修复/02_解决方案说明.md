## 根因

段落索引在写入端（ReaderPage markdown AST 渲染）和读取端（`calcProgress` 纯文本 split）使用了完全不同的段落模型。多章节文章进一步放大了这一矛盾（节内索引 vs 全文索引）。

## 一、Fix 1+2：统一段落解析 + 多章节进度（P0）

### 1.1 新增：`src/services/progress.js`

创建统一的段落解析和进度计算模块，作为 ReaderPage 和 ImportPage 的共同数据源。

```js
/**
 * 段落解析与阅读进度计算
 *
 * 此模块定义了「段落」的唯一权威来源。
 * 写入端（ReaderPage）和读取端（ImportPage/calcProgress）均使用此模块的函数。
 */

/**
 * 将文章解析为段落文本数组
 *
 * 段落定义规则（与 ReaderPage MarkdownContent 的 paraIdxRef 递增逻辑一致）:
 *   - markdown 文章: 去除 frontmatter → 按 \n\n+ 分割为块 → 标题块跳过 →
 *     列表块拆分为单项 → 其余块作为独立段落
 *   - 纯文本文章: 按 \n+ 分割（与 parseText() 一致）
 */
export function getParagraphs(article) {
  // !! 而非 != null：防止空字符串误走 markdown 路径（空 markdown 无段落结构）
  if (!!article.markdown) {
    return getMarkdownParagraphs(article.markdown)
  }
  return article.text
    .split(/\n+/)
    .map(p => p.trim())
    .filter(p => p.length > 0)
}

function getMarkdownParagraphs(markdown) {
  // 去除 YAML frontmatter
  const body = markdown.replace(/^---\r?\n[\s\S]*?\r?\n---\r?\n?/, '')

  // 提取围栏代码块（```...```），用占位符替换，防止内部空行被 \n\n+ 错误分裂
  const codeBlocks = []
  const bodyWithoutCode = body.replace(/```[\s\S]*?```/g, (match) => {
    codeBlocks.push(match)
    return `%%CODEBLOCK_${codeBlocks.length - 1}%%`
  })

  // 按空行分割为块
  const blocks = bodyWithoutCode.split(/\n\n+/).map(b => b.trim()).filter(b => b.length > 0)

  const paragraphs = []
  for (const block of blocks) {
    // 跳过代码块占位符（代码块在 MarkdownContent 中不递增 paraIdxRef）
    if (/^%%CODEBLOCK_\d+%%$/.test(block)) continue
    // 跳过 h1-h6 标题行（所有级别标题在 MarkdownContent 中均不递增 paraIdxRef）
    if (/^#{1,6}\s/.test(block)) continue
    // 跳过水平线（---, ***, ___ 等）
    if (/^[-*_]{3,}\s*$/.test(block)) continue

    // ⚠️ 引用块处理已修正（2026-06-20）：
    // 原逻辑用 /^>\s/m 跳过引用块，但 MarkdownContent 对引用块内部 <p>/<li> 递增 paraIdxRef，
    // 导致段落计数不一致 → paragraphIndex 越界 → 99% 假阳性。
    // 修正方案：在 \n\n+ 分割前用 bodyNormalized 去除 > 前缀（见下方 §补充修复）。
    // 此处不再跳过引用块——bodyNormalized 已将引用块内部空行还原，段落自然分割。

    // 列表块：拆分为单项（每行一个 list item）
    if (isListBlock(block)) {
      const items = block.split(/\n/).map(l => l.trim()).filter(l => l.length > 0)
      for (const item of items) {
        paragraphs.push(cleanInlineMarkdown(item.replace(/^[-*+]\s+|\d+\.\s+/, '')))
      }
    } else {
      paragraphs.push(cleanInlineMarkdown(block))
    }
  }
  return paragraphs
}

function isListBlock(block) {
  const lines = block.split(/\n/)
  return lines.every(l => /^\s*[-*+]\s|^\s*\d+\.\s/.test(l.trim()))
}

/** 去除行内 markdown 标记，保留纯文本用于词数统计 */
function cleanInlineMarkdown(text) {
  return text
    .replace(/!\[.*?\]\(.*?\)/g, '')
    .replace(/\[([^\]]*)\]\(.*?\)/g, '$1')
    .replace(/`{1,3}[^`]*`{1,3}/g, '')
    .replace(/[*_~]{1,3}/g, '')
    .replace(/^>\s?/gm, '')
    .trim()
}

/** 统计词数 */
function countWords(text) {
  return text.split(/\s+/).filter(Boolean).length
}

/** 获取段落数组中指定索引处的累积词数 */
export function getWordOffsetAtParagraph(article, paragraphIndex) {
  const paragraphs = getParagraphs(article)
  if (paragraphIndex < 0 || paragraphIndex >= paragraphs.length) {
    return countWords(paragraphs.join(' '))
  }
  return countWords(paragraphs.slice(0, paragraphIndex + 1).join(' '))
}

/** 获取文章总词数 */
export function getTotalWords(article) {
  return article.wordCount || countWords(article.text)
}

// ═════════════════════════════════════════════════════════════════════
// 进度计算（兼容单章节和多章节文章）
// ═════════════════════════════════════════════════════════════════════

/**
 * 计算阅读进度百分比
 *
 * @param {Object} article — { sections, sectionCount, wordCount, text, markdown }
 * @param {Object|null} mark  — { paragraphIndex, completed, sectionId, completedSections }
 * @returns {number|null} 0-99 的进度百分比，或 null（无进度）
 */
export function calcProgress(article, mark) {
  if (!mark || mark.completed) return null
  if (mark.paragraphIndex === null || mark.paragraphIndex === undefined) return null

  // 多章节文章：按 section 词数汇总
  if (article.sectionCount > 1 && Array.isArray(article.sections)) {
    return calcMultiSectionProgress(article, mark)
  }

  return calcSingleProgress(article, mark.paragraphIndex)
}

function calcSingleProgress(article, paragraphIndex) {
  const totalWords = getTotalWords(article)
  if (totalWords === 0) return null
  const readWords = getWordOffsetAtParagraph(article, paragraphIndex)
  return Math.min(99, Math.round((readWords / totalWords) * 100))
}

function calcMultiSectionProgress(article, mark) {
  const totalWords = getTotalWords(article)
  if (totalWords === 0) return null

  const completedIds = new Set(mark.completedSections ?? [])
  let cumulativeWords = 0

  for (const section of article.sections) {
    if (completedIds.has(section.id)) {
      // 已完成章节：计入全部词数
      cumulativeWords += section.body.wordCount
    } else if (section.id === mark.sectionId) {
      // 当前章节：按节内段落索引计算已读词数
      const sectionArticle = {
        text: section.body.text,
        markdown: section.body.markdown,
        wordCount: section.body.wordCount,
      }
      cumulativeWords += getWordOffsetAtParagraph(sectionArticle, mark.paragraphIndex)
      break
    }
    // 其他未完成章节不计入
  }

  return Math.min(99, Math.round((cumulativeWords / totalWords) * 100))
}
```

## 补充修复（2026-06-20）：引用块段落计数对齐

**关联分析**：[2026-06-20/01_阅读进度99pct异常](../../2026-06-20/01_阅读进度99pct异常/修复/01_问题分析报告.md)

### 问题

`getMarkdownParagraphs` 跳过引用块（`/^>\s/m`），但 `MarkdownContent` 对引用块内部 `<p>`/`<li>` 递增 `paraIdxRef`。含引用块的文章中，AST 段落索引可能超出 regex 段落数组长度 → `getWordOffsetAtParagraph` 溢出 → 全部词数 → 99%。

### 修改：`getMarkdownParagraphs` — 分割前去除引用标记

```diff
 function getMarkdownParagraphs(markdown) {
   const body = markdown.replace(/^---\r?\n[\s\S]*?\r?\n---\r?\n?/, '')

+  // 去除行首引用标记（> ），将引用块内空白行还原为真正的空行，
+  // 使 \n\n+ 分割能正确拆分引用块内的多个段落
+  const bodyNormalized = body.replace(/^>\s?/gm, '')
+
   const codeBlocks = []
-  const bodyWithoutCode = body.replace(/```[\s\S]*?```/g, (match) => {
+  const bodyWithoutCode = bodyNormalized.replace(/```[\s\S]*?```/g, (match) => {
     ...
   })
```

同时移除循环中的 `if (/^>\s/m.test(block)) continue`（见上方 §1.1 注释）。

### 有效性

引用块内部段落现在被正确计入，`getMarkdownParagraphs` 与 `MarkdownContent` 的 `paraIdxRef` 计数一致 → 不再溢出 → 进度百分比反映真实阅读位置。
